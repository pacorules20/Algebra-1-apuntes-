\section*{Divisibilidad}

\begin{definicion}
Sean $a,b \in R$. Decimos que $a$ es divisible a $b$ si hay otro entero
$c$ tal que $b=ac$, y en ese caso escribimos $a\mid b$. Si $a$ no divide
a $b$ escribimos $a \nmid b$.
\end{definicion}

\[
R = \{ (a,b) \in \mathbb{Z} \times \mathbb{Z} : b \neq 0 \}
\]
es una relación en $\mathbb{Z}$.

Sea $a \in \mathbb{Z}$. Como $a=a*1$, $a\mid a$, así que $(a,a) \in R$ y
$R$ es reflexiva. $R$ no es simétrica porque $(3,9) \in R$ pero
$(9,3) \notin R$. Falta ver que $(a,b),(b,c) \in R \rightarrow (a,c) \in R$.

Supongamos que $(a,b),(b,c) \in R$, entonces hay enteros $x,y$ tales que
$b=ax$ y $c=by$, entonces $c=by=axy$, así que $(a,c) \in R$ y $R$ es
transitiva. Así que $a\mid c$ y $(a,c) \in R$.

\textbf{Antisimétrica:} $(a,b) \in R$, $(b,a) \in R \rightarrow a=b$.

Hay $x$ tal que $b=ax$. Hay $y$ tal que $a=by$.

Contraejemplo: $a=1$, $b=-1$.

Si $(a,b),(b,a) \in R$, entonces $a=b$ o $a=-b$.

\begin{teoria}
Si $(a,b),(b,a) \in R$, de manera que hay enteros $x$ e $y$ tales que
$b=ax$ y $a=by$, tal que
\[
a = by = (ax)y = a(xy)
\]
así que
\[
0 = a - a(xy) = a(1-xy)
\]
y por lo tanto $a=0$ o $1-x=0$.

Supongamos que $a=0$, entonces $b=ax=0x=0$.

Supongamos que $a \neq 0$, así que $xy=1$, $x$ es invertible, así que
$x=1$ o $x=-1$. Si $x=1$ entonces $b=ax=a$, y si $x=-1$ entonces
$b=ax=a(-1)=-a$.

$R$ no es antisimétrica.
\end{teoria}

\begin{teoria}
La relación
\[
R' = \{ (a,b) \in \text{No} \times \text{No} : a \mid b \}
\]
es una relación de orden en $\text{No}$.

$(a,b),(b,a) \in R' \rightarrow a=b$ (no se puede $a=-b$).
\end{teoria}
