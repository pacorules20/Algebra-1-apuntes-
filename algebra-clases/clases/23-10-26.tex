\section{Raíces en $\mathbb C$: repaso}
\begin{proposicion}
Sea $z=re^{i\theta}\in\mathbb C$ no nulo, con $r>0$, y sea $n\geq1$
un entero. Existen exactamente $n$ raíces $n$-ésimas distintas de $z$:
\[
w_k=\sqrt[n]{r}\,e^{i\theta/n}e^{2\pi ik/n}
=\sqrt[n]{r}\,e^{i(\theta+2\pi k)/n},\qquad k=0,\dots,n-1.
\]
Equivalentemente, el polinomio $X^n-z$ tiene exactamente $n$ raíces
distintas en $\mathbb C$.
\end{proposicion}
\begin{demostracion}
Cada número indicado satisface $w_k^n=re^{i\theta}=z$.
Recíprocamente, si $w=\rho e^{i\varphi}$ y $w^n=z$, entonces
$\rho^n=r$ y $n\varphi=\theta+2\pi k$ para algún $k\in\mathbb Z$.
La fórmula anterior describe, por lo tanto, todas las raíces.
Además, $w_k=w_j$ equivale a $n\mid(k-j)$, lo cual para
$0\leq k,j<n$ equivale a $k=j$.
\end{demostracion}
\begin{nota}{Notación}
El radical $\sqrt[n]{r}$ es la raíz real positiva de $r$; no se trata
de una raíz cuadrada salvo cuando $n=2$. Si $z=0$, el polinomio
$X^n$ tiene una única raíz distinta, $0$, de multiplicidad $n$.
\end{nota}

\section{Polinomios sobre un cuerpo}
En adelante, $K$ es un cuerpo y $K[X]$ es el conjunto de polinomios
con coeficientes en $K$. Usamos $X$ como indeterminada.

\begin{definicion}
Un polinomio es una suma formal finita
\[
f=\sum_{i=0}^{d}a_iX^i,\qquad a_i\in K.
\]
Dos polinomios son iguales si y solo si sus coeficientes de cada
potencia de $X$ son iguales, completando con ceros cuando sea necesario.
Si $f\neq0$, su \textbf{grado}, denotado $\deg f$, es el mayor índice
$i$ para el cual $a_i\neq0$; ese $a_i$ es su \textbf{coeficiente principal}.
\end{definicion}
\begin{nota}{Grado del polinomio cero}
El polinomio cero tiene todos sus coeficientes nulos y no tiene grado
como entero no negativo. Puede adoptarse la convención
$\deg0=-\infty$. En particular, el polinomio cero no tiene grado $0$:
los polinomios de grado $0$ son las constantes no nulas.
\end{nota}
\begin{nota}{Polinomios y funciones}
La igualdad formal se refiere a coeficientes. Sobre un cuerpo finito,
dos polinomios distintos pueden definir la misma función. Por ejemplo,
sobre $\mathbb F_2$, los polinomios $X^2$ y $X$ coinciden al evaluarlos
en cada elemento del cuerpo, pero no son iguales en $\mathbb F_2[X]$.
\end{nota}

\subsection{Suma y grado}
Si $f=\sum_{i=0}^{d}a_iX^i$ y $g=\sum_{i=0}^{e}b_iX^i$, tomando
$N=\max\{d,e\}$ y completando con coeficientes nulos, definimos
\[
f+g=\sum_{i=0}^{N}(a_i+b_i)X^i.
\]
\begin{proposicion}
Si $f,g$ son no nulos y $f+g\neq0$, entonces
\[
\deg(f+g)\leq\max\{\deg f,\deg g\}.
\]
Hay igualdad si $\deg f\neq\deg g$. Si ambos tienen grado $d$, hay
igualdad si y solo si $a_d+b_d\neq0$.
\end{proposicion}
\begin{demostracion}
No aparecen términos de grado superior al máximo de los grados.
Si los grados son distintos, el término de mayor grado de la suma
proviene de un solo sumando y no puede cancelarse. Si son iguales,
el coeficiente de $X^d$ en la suma es $a_d+b_d$.
\end{demostracion}
\begin{ejemplo}
En $\mathbb Q[X]$, si $f=X^2+1$ y $g=-X^2+X$, entonces $f+g=X+1$:
el grado baja porque se cancelan los coeficientes principales.
\end{ejemplo}

\subsection{Producto y grado}
El producto se obtiene distribuyendo y agrupando potencias iguales:
\[
fg=\left(\sum_{i=0}^{d}a_iX^i\right)
\left(\sum_{j=0}^{e}b_jX^j\right)
=\sum_{\ell=0}^{d+e}\left(
\sum_{\substack{0\leq i\leq d,\ 0\leq j\leq e\\i+j=\ell}}
a_ib_j\right)X^\ell.
\]
\begin{proposicion}
Si $f,g\neq0$, entonces $fg\neq0$ y
\[
\deg(fg)=\deg f+\deg g.
\]
En consecuencia, $fg=0$ si y solo si $f=0$ o $g=0$.
\end{proposicion}
\begin{demostracion}
Si $\deg f=d$ y $\deg g=e$, el coeficiente de $X^{d+e}$ en $fg$ es
$a_db_e$. Como $K$ es un cuerpo y ambos factores son no nulos,
$a_db_e\neq0$. No hay términos de grado mayor que $d+e$.
Si algún factor es cero, el producto es cero; el recíproco se sigue
del caso de factores no nulos.
\end{demostracion}
Así, con estas operaciones, $K[X]$ es un anillo conmutativo con unidad
$1$ y sin divisores de cero: un \defterm{dominio de integridad}.

\section{Elementos inversibles y divisibilidad}
\begin{proposicion}
Los elementos inversibles de $K[X]$ son exactamente las constantes
no nulas, es decir, los elementos de $K^\times=K\setminus\{0\}$.
En particular, $K[X]$ no es un cuerpo.
\end{proposicion}
\begin{demostracion}
Si $c\in K$ es no nulo, su inverso $c^{-1}\in K$ también pertenece a
$K[X]$. Si $fg=1$, ambos factores son no nulos y
\[
0=\deg1=\deg(fg)=\deg f+\deg g.
\]
Como ambos grados son no negativos, los dos son $0$. Por lo tanto,
$f$ y $g$ son constantes no nulas. El polinomio $X$, por ejemplo,
no es inversible, aunque es no nulo.
\end{demostracion}
\begin{ejemplo}
En $\mathbb Q[X]$, una constante $q\in\mathbb Q\setminus\{0\}$ tiene
inverso $1/q$. En cambio, $X$ y $X+1$ no tienen inverso en $\mathbb Q[X]$.
\end{ejemplo}

\begin{definicion}[{title={Divisibilidad de polinomios}}]
Dados $f,g\in K[X]$, decimos que $f$ \textbf{divide} a $g$, y escribimos
$f\mid g$, si existe $h\in K[X]$ tal que $g=fh$.
\end{definicion}
\begin{proposicion}
Si $f,g\in K[X]$ son no nulos, entonces
\[
f\mid g\ \text{y}\ g\mid f
\quad\Longleftrightarrow\quad
 g=cf\ \text{para algún }c\in K^\times.
\]
En este caso se dice que $f$ y $g$ son \textbf{asociados}.
\end{proposicion}
\begin{demostracion}
Si $g=fu$ y $f=gv$, entonces $g=gvu$ y $g(vu-1)=0$.
Como $g\neq0$ y no hay divisores de cero, $vu=1$.
Así, $u,v$ son constantes no nulas y $g=uf$.
Recíprocamente, si $g=cf$ con $c\neq0$, entonces $f=c^{-1}g$,
por lo que se cumplen ambas divisibilidades.
\end{demostracion}

\section{Congruencia e irreducibilidad}
\begin{definicion}[{title={Congruencia módulo un polinomio}}]
Fijado $f\in K[X]$ no nulo, decimos que $u,v\in K[X]$ son
\textbf{congruentes módulo $f$} si $f\mid(u-v)$. Escribimos
\[
u\equiv v\pmod f.
\]
\end{definicion}
No es necesario que $u$ y $v$ sean no constantes. Si $f$ es una
constante no nula, todos los polinomios son congruentes entre sí.
La congruencia es una relación de equivalencia y es compatible con
la suma y el producto: si $u\equiv v\pmod f$ y $s\equiv t\pmod f$,
entonces $u+s\equiv v+t\pmod f$ y $us\equiv vt\pmod f$.
En efecto, $(u+s)-(v+t)=(u-v)+(s-t)$ y
$us-vt=s(u-v)+v(s-t)$ son múltiplos de $f$.

\begin{definicion}[{title={Polinomio irreducible}}]
Un polinomio $p\in K[X]$ es \textbf{irreducible} si es no nulo, no
inversible y, para toda factorización $p=uv$ con $u,v\in K[X]$, alguno
de los factores es inversible.
\end{definicion}
Equivalentemente, $\deg p\geq1$ y $p$ no puede escribirse como producto
de dos polinomios de grado positivo.
\begin{ejemplo}
Todo polinomio de grado $1$ es irreducible. La irreducibilidad depende
del cuerpo: $X^2+1$ es irreducible en $\mathbb R[X]$, pero en
$\mathbb C[X]$ se factoriza como $(X-i)(X+i)$.
\end{ejemplo}

\section{Algoritmo de la división de polinomios}
\begin{proposicion}[{title={Algoritmo de la división}}]
Sea $K$ un cuerpo. Dados $g,f\in K[X]$, con $f\neq0$, existen únicos
polinomios $q,r\in K[X]$ tales que
\[
g=qf+r,\qquad r=0\ \text{o}\ \deg r<\deg f.
\]
El polinomio $g$ es el dividendo, $f$ el divisor, $q$ el cociente y
$r$ el resto.
\end{proposicion}
\begin{demostracion}
\textbf{Existencia.} Si $g=0$, tomamos $q=r=0$.
Si $\deg f=0$, el divisor es una constante no nula y podemos tomar
$q=f^{-1}g$ y $r=0$. Supongamos entonces $m=\deg f\geq1$.

Hacemos inducción fuerte sobre $n=\deg g$ para dividendos no nulos.
Si $n=0$, tomamos $q=0$ y $r=g$, pues $0=\deg g<m$.
Supongamos que el resultado vale para todos los grados menores que
$n$ y consideremos $g$ de grado $n$. Si $n<m$, nuevamente $q=0$ y
$r=g$ sirven. Si $n\geq m$, escribimos los términos principales como
$a_nX^n$ y $b_mX^m$, con $a_n,b_m\neq0$, y definimos
\[
t=\frac{a_n}{b_m}X^{n-m},\qquad h=g-tf.
\]
Los términos de grado $n$ se cancelan, por lo que $h=0$ o
$\deg h<n$. Si $h=0$, tomamos $q=t$ y $r=0$.
Si $h\neq0$, la hipótesis inductiva da $h=q_1f+r_1$, con
$r_1=0$ o $\deg r_1<m$. Entonces
\[
g=tf+h=(t+q_1)f+r_1.
\]
Así, $q=t+q_1$ y $r=r_1$ satisfacen lo requerido.

\textbf{Unicidad.} Supongamos que
$g=qf+r=q'f+r'$, con ambos restos admisibles. Entonces
\[
(q-q')f=r'-r.
\]
Si $q\neq q'$, el miembro izquierdo es no nulo y tiene grado
$\deg(q-q')+\deg f\geq\deg f$. El miembro derecho, si es no nulo,
tiene grado menor que $\deg f$; si es cero, tampoco puede ser igual
al miembro izquierdo no nulo. Esto es una contradicción.
Luego $q=q'$ y, sustituyendo, $r=r'$.
\end{demostracion}
\begin{nota}{Por qué se requiere un cuerpo}
El paso que elimina el término principal utiliza $a_n/b_m$.
En un cuerpo todo coeficiente $b_m\neq0$ tiene inverso.
Sobre un anillo arbitrario este paso puede fallar; por ejemplo, en
$\mathbb Z[X]$ no se puede dividir $X$ por $2X$ con resto cero
o de grado menor que $1$ y cociente de coeficientes enteros.
\end{nota}
\begin{ejemplo}
En $\mathbb Q[X]$, al dividir $g=X^3+1$ por $f=X^2+1$ resulta
\[
X^3+1=X(X^2+1)+(1-X).
\]
Por lo tanto $q=X$ y $r=1-X$, con $\deg r=1<2=\deg f$.
En particular, $X^3+1\equiv1-X\pmod{X^2+1}$.
\end{ejemplo}
