\section{La ecuación de congruencia $ax\equiv b \pmod m$}

\begin{proposicion}
Sean $a,b,m\in\mathbb{Z}$ con $m\ge 1$, y sea $d=(a:m)$. La ecuación
\[
ax\equiv b \pmod m
\]
tiene solución si y sólo si $d\mid b$.

En ese caso, si $u,v\in\mathbb{Z}$ cumplen $ua+vm=d$ (Bézout) y
$x_0=\dfrac{ub}{d}$, entonces $x_0$ es una solución y el conjunto de todas las
soluciones es una única clase de congruencia módulo $m/d$:
\[
[x_0]=\left\{x_0+k\frac{m}{d} : k\in\mathbb{Z}\right\}
=\left\{x\in\mathbb{Z} : x\equiv x_0 \pmod{\tfrac{m}{d}}\right\}.
\]
Módulo $m$ son exactamente $d$ clases.
\end{proposicion}

\begin{demostracion}
Tenemos
\[
ax\equiv b \pmod m \iff m\mid ax-b \iff ax-b=mk \text{ para algún }k\in\mathbb{Z}
\iff ax+m(-k)=b,
\]
que es una ecuación diofántica en $(x,-k)$ con coeficientes $a$ y $m$, cuyo
mcd es $d$. Por la proposición del 9-10-26, tiene solución si y sólo si $d\mid b$.

En ese caso, de $ua+vm=d$ resulta
\[
a\cdot\frac{ub}{d}+m\cdot\left(-\frac{vb}{d}\right)=\frac{b(ua+vm)}{d}=b,
\]
así que $x_0=ub/d$ es solución. Por la fórmula general, los $x$ de las soluciones
son $x=x_0+k\,\tfrac{m}{d}$ con $k\in\mathbb{Z}$.
\end{demostracion}

\begin{proposicion}
Si $(a:m)=1$ y $u\in\mathbb{Z}$ cumple $ua\equiv 1 \pmod m$, entonces
\[
ax\equiv b \pmod m \iff x\equiv ub \pmod m,
\]
es decir, $\{x\in\mathbb{Z} : ax\equiv b \pmod m\}=\{x\in\mathbb{Z} : x\equiv ub \pmod m\}$.
\end{proposicion}

\begin{demostracion}
($\Rightarrow$) Si $ax\equiv b$, multiplicando por $u$:
$x\equiv uax\equiv ub \pmod m$.

($\Leftarrow$) Si $x\equiv ub$, multiplicando por $a$: $ax\equiv aub\equiv 1\cdot b=b \pmod m$.
\end{demostracion}

\begin{ejemplo}
Resolver $4x\equiv 4 \pmod 8$. Tenemos $d=(4:8)=4\mid 4$, así que hay solución.
Dividiendo todo por $4$ (Regla 1): $x\equiv 1 \pmod 2$. Luego las soluciones son
los impares, y módulo $8$ son $d=4$ clases: $[1],[3],[5],[7]$.
\end{ejemplo}

\begin{nota}
\textbf{Método general para $ax\equiv b \pmod m$.}
\begin{enumerate}
    \item Calcular $d=(a:m)$ y verificar que $d\mid b$ (si no, no hay solución).
    \item Simplificar: $a_1=a/d$, $b_1=b/d$, $m_1=m/d$. Entonces
    $a_1x\equiv b_1 \pmod{m_1}$ tiene las mismas soluciones (Regla 1).
    \item Como $(a_1:m_1)=1$, buscar $u$ con $a_1u\equiv 1 \pmod{m_1}$. Entonces las
    soluciones son $x\equiv ub_1 \pmod{m_1}$.
\end{enumerate}
\end{nota}

\section{Sistemas de dos congruencias con módulos no coprimos}

\begin{proposicion}
Sean $n_1,n_2\ge 1$ y $g=(n_1:n_2)$. El sistema
\[
\begin{cases}
x\equiv c_1 \pmod{n_1}\\
x\equiv c_2 \pmod{n_2}
\end{cases}
\]
tiene solución si y sólo si $g\mid c_2-c_1$. En ese caso, el conjunto de
soluciones es una única clase módulo $[n_1:n_2]=\dfrac{n_1n_2}{g}$ (el mínimo
común múltiplo).
\end{proposicion}

\begin{demostracion}
($\Rightarrow$) Si $x$ es solución, $n_1\mid x-c_1$ y $n_2\mid x-c_2$, así que
$g$ divide a ambos, y entonces $g\mid (x-c_1)-(x-c_2)=c_2-c_1$.

($\Leftarrow$) Las soluciones de la primera ecuación son $x=c_1+n_1k$ con
$k\in\mathbb{Z}$. Reemplazando en la segunda:
\[
c_1+n_1k\equiv c_2 \pmod{n_2} \iff n_1k\equiv c_2-c_1 \pmod{n_2}.
\]
Como $(n_1:n_2)=g\mid c_2-c_1$, esta ecuación tiene solución, y sus soluciones
son una clase $k\equiv k_0 \pmod{n_2/g}$. Entonces $k=k_0+\frac{n_2}{g}j$ y
\[
x=c_1+n_1k_0+\frac{n_1n_2}{g}\,j, \qquad j\in\mathbb{Z},
\]
o sea $x\equiv c_1+n_1k_0 \pmod{n_1n_2/g}$.
\end{demostracion}

\begin{nota}
Si $g=1$ es el teorema chino del resto. Si $g>1$ puede no haber solución: la
condición $g\mid c_2-c_1$ es la \emph{compatibilidad} de las dos ecuaciones.
\end{nota}

\begin{nota}
\textbf{Método para un sistema $a_ix\equiv b_i \pmod{m_i}$.} Primero se reduce
cada ecuación a la forma $x\equiv c_i \pmod{n_i}$ (verificando que $(a_i:m_i)\mid b_i$),
y después se juntan de a dos con la proposición anterior.
\end{nota}

\begin{ejemplo}
Resolver
\[
\begin{cases}
6x\equiv 6 \pmod{15}\\
35x\equiv 14 \pmod{21}.
\end{cases}
\]
\textbf{Primera ecuación.} $(6:15)=3\mid 6$. Dividiendo por $3$:
$2x\equiv 2 \pmod 5$, y como $(2:5)=1$ queda $x\equiv 1 \pmod 5$.

\textbf{Segunda ecuación.} $(35:21)=7\mid 14$. Dividiendo por $7$:
$5x\equiv 2 \pmod 3$, o sea $2x\equiv 2 \pmod 3$, y queda $x\equiv 1 \pmod 3$.

\textbf{Juntando.} $x\equiv 1 \pmod 5$ y $x\equiv 1 \pmod 3$, con $(5:3)=1$:
por el teorema chino del resto $x\equiv 1 \pmod{15}$.

\textbf{Chequeo.} $x=1$: $6\equiv 6 \pmod{15}$ y $35-14=21\equiv 0 \pmod{21}$. $\checkmark$
\end{ejemplo}

\begin{ejemplo}
Módulos no coprimos. Resolver $x\equiv 1 \pmod 4$ y $x\equiv 3 \pmod 6$.

Acá $g=(4:6)=2$ y $c_2-c_1=2$, que es divisible por $2$: hay solución. Escribimos
$x=1+4k$ y reemplazamos: $4k\equiv 2 \pmod 6 \iff 2k\equiv 1 \pmod 3 \iff k\equiv 2 \pmod 3$.
Entonces $x=1+4\cdot 2=9$, y la solución es $x\equiv 9 \pmod{12}$, pues
$[4:6]=12$.

En cambio, $x\equiv 1 \pmod 4$ y $x\equiv 2 \pmod 6$ no tiene solución, porque
$g=2\nmid c_2-c_1=1$.
\end{ejemplo}

\section{Fermat y reciprocidad cuadrática}

\begin{nota}
\textbf{Pequeño teorema de Fermat.} Si $p$ es primo y $p\nmid a$, entonces
$a^{p-1}\equiv 1 \pmod p$. Por lo tanto, $a^n\equiv a^{r_{p-1}(n)} \pmod p$ para
todo $n\in\mathbb{N}_0$ (ver la sección del pequeño teorema de Fermat).
\end{nota}

\begin{ejemplo}
Calcular el resto de dividir $2^{1000}$ por $7$.

Como $7$ es primo y $7\nmid 2$, por Fermat $2^6\equiv 1 \pmod 7$. Dividiendo,
$1000=6\cdot 166+4$, así que $r_6(1000)=4$ y
\[
2^{1000}\equiv 2^4=16\equiv 2 \pmod 7.
\]
El resto es $2$.
\end{ejemplo}

\begin{nota}
\textbf{Ley de reciprocidad cuadrática} (enunciado, sin demostración). Sean $p\neq q$
primos impares. Las ecuaciones
\[
x^2\equiv p \pmod q \qquad\text{y}\qquad x^2\equiv q \pmod p
\]
están relacionadas: si $p\equiv 1 \pmod 4$ o $q\equiv 1 \pmod 4$, ambas tienen
solución o ambas no la tienen; si $p\equiv q\equiv 3 \pmod 4$, exactamente una de
las dos tiene solución.
\end{nota}
