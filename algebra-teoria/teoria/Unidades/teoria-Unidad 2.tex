\chapter{Numeros naturales e Induccion}
\begin{definicion}[title= Definicion de sucesion]
    Una sucesion es una lista de ordenada de elementos que siguen una regla de " patron"
\end{definicion}
\begin{definicion}[title= Definicion de serie]
    Una serie es la suma de todos los elementos de una sucesion infinita
\end{definicion}

\begin{definicion}[title= Definicion de serie geometrica]
    Es la serie que toma las n+1 primeras potencias de un numero (q) \\
    tiene si aplicamos la logica de la suma de gauss nos queda algo de la forma:
    \[
    Q = \frac{q^{n+1}-1}{q-1}, si \hspace{0.3 cm} q\neq 1     \]
    \[
    Q = n+1, si \hspace{0.3cm} q=1 
    \]
\end{definicion}

\textbf{Sucesion Fibonnaci}
\begin{definicion}
    son las sucesiones de la forma:
    \[  
    f0 = 0 \hspace{0.3 cm}, f1 = 1 \hspace{0.3 cm} , fn+2 = fn+1 + fn
    \]
\end{definicion}
