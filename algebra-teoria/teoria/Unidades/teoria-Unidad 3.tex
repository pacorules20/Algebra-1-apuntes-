\section{Cardinal de conjuntos y cantidad de relaciones}

%\subsection{Principios de cardinal}
\begin{definicion}[title=Definion de cardinal de un conjunto]
    \textbf{sea A un conjunto, se llama cardinal de A a la cantidad de elementos distintos que tiene A, y se anota \#A. }
\end{definicion}
\begin{ejemplo}
    \# $\emptyset$ = 0 , \#\{a, b, c\}= 3  
\end{ejemplo}

\begin{teoria}[title=Propiedades de los cardinales]
    \textbf{sea A y B dos conjuntos} 
    \begin{listapropiedades}
  \item[1] Sea A un Conjuntos finitos y sea $B \subseteq A $ entonces $\#B \leq \#A $
  \item[2] $(A\cup B)$ = $\#A + \#B - \#(A \cup B)$ 
  \item[3] $\#\mu < \infty \rightarrow \#A^c = \#\mu - \#A$ 
  \item[4] $\#(A-B) = \#A - \#(A\cap B) $
  \item[5] $\# (A\Delta B) = \#A - \#B -2\#(A\cap B)$ 
  \item[6] $\# (A X B) = \#A * \#B $
  \item[7] $P(A) = 2^{\#A}$
    \end{listapropiedades}
\end{teoria}

\begin{teoria}[title= Cantidad de Funcion]
    \textbf{Sean A y B conjuntos finitos}
        \begin{listapropiedades}
  \item[1] $\#(f: A\rightarrow B)$, f es funcion. Entonces $\#B^{\#A}$
  \item[2] $\#(f: A\rightarrow B)$ f es una funcion Biyectiva. Entonces $n!$
  \item[3] $\#(f: A\rightarrow B)$ f es una funcion Inyectiva. Entonces $\frac{k!}{(k-n)!}$
    \end{listapropiedades}
\end{teoria}


\begin{teoria}[title= Cardinales y Funciones]
    \textbf{Sean A y B conjuntos finitos}
        \begin{listapropiedades}
  \item[1] Sea $f: A\rightarrow B$ es una funcion Inyectiva. Entonces $\#A \leq \#B$
  \item[2] Sea $f: A\rightarrow B$ es una funcion Sobreyectiva. Entonces $\#A \geq \#B$
  \item[3] Sea $f: A\rightarrow B$ es una funcion Biyectiva. Entonces $\#A = \#B$
    \end{listapropiedades}
\end{teoria}

\section{El factorial}

\begin{definicion}[title={Definición de Factorial}]
    Sea $n \in \mathbb{N}$. El factorial de $n$, que se denota como $n!$, es el número natural definido como:
    \[
    n! = n \cdot (n-1) \cdot \dots \cdot 2 \cdot 1 = \prod_{i=1}^{n} i
    \]
\end{definicion}

\begin{proposicion}[title=Factorial y permutaciones]
    La cantidad de funciones biyectivas entre dos conjuntos de $n$ elementos, es decir, la cantidad de permutaciones de un conjunto de $n$ elementos, es igual a $n!$.
\end{proposicion}

\begin{demostracion}
    Es un caso particular de la proposición de funciones inyectivas, tomando $m=n$: la cantidad de funciones inyectivas de un conjunto de $n$ elementos en otro de $n$ elementos es
    \[
    \frac{n!}{(n-n)!} = \frac{n!}{0!} = n!
    \]
    Notar que en este caso ($m=n$), toda función inyectiva es también biyectiva (y viceversa), ya que ambos conjuntos tienen la misma cantidad de elementos.
\end{demostracion}

\begin{proposicion}[title= Cantidad de funciones inyectivas]
    Sean $A_m$ y $B_n$ conjuntos finitos con $m$ y $n$ elementos respectivamente, donde $m \leq n$.
    La cantidad de funciones inyectivas de $A_m$ en $B_n$ es:
    \[
        \frac{n!}{(n-m)!}
    \]
\end{proposicion}

\begin{demostracion}
    Para formar una lista ordenada de $m$ elementos extraídos de un conjunto de $n$ elementos, tenemos $n$ opciones para el primer elemento, $n-1$ para el segundo, y así sucesivamente, obteniendo un total de:
    \[
    n(n-1)(n-2)\cdots(n-m+1) = \frac{n!}{(n-m)!}
    \]
\end{demostracion}

\section{El Número Combinatorio}

\begin{proposicion}[title=Teorema: Número Combinatorio]
    Sea $n \in \mathbb{N}_0$ y sea $A_n$ un conjunto con $n$ elementos. Para $0 \leq k \leq n$, se nota con el símbolo $\binom{n}{k}$, que se llama \textbf{número combinatorio}, y representa la cantidad de subconjuntos con $k$ elementos de un conjunto de $n$ elementos, sin importar el orden interno de cada subconjunto, y es igual a:
    \[
    \binom{n}{k} = \frac{n!}{k!(n-k)!}
    \]
\end{proposicion}

\begin{demostracion}
    Contamos primero cuántas listas ordenadas de $k$ elementos distintos se pueden formar a partir de un conjunto de $n$ elementos. Por la proposición anterior (con $m=k$), esa cantidad de formas ordenadas (variaciones simples) es:
    \[
    \frac{n!}{(n-k)!}
    \]
    Como el orden interno de esos $k$ elementos no nos importa, y cada subconjunto de $k$ elementos puede ordenarse de $k!$ maneras distintas, dividimos el total anterior por $k!$ para eliminar las repeticiones debidas al orden. De este modo, la cantidad de subconjuntos sin importar el orden es:
    \[
    \binom{n}{k} = \frac{n!}{k!(n-k)!}
    \]
\end{demostracion}


\begin{definicion}[title=Triángulo de Pascal]
    El \textbf{triángulo de Pascal} es un arreglo triangular de números en el cual la fila $n$ (para $n \in \mathbb{N}_0$) está formada por los números combinatorios:
    \[
    \binom{n}{0}, \binom{n}{1}, \binom{n}{2}, \dots, \binom{n}{n}
    \]
\end{definicion}

\begin{teoria}[title= Identidad de Stifel]
    Sea $n \in \mathbb{N}$ y $1 \leq k \leq n-1$. Entonces:
    \[
    \binom{n-1}{k-1} + \binom{n-1}{k} = \binom{n}{k}
    \]
    Esta identidad muestra que cada número del triángulo de Pascal (salvo los bordes) es la suma de los dos números que están inmediatamente arriba en la fila anterior.
\end{teoria}

\begin{demostracion}
    Partimos de la expresión inicial aplicando la definición de número combinatorio:
    \[
    \binom{n-1}{k-1} + \binom{n-1}{k} = \frac{(n-1)!}{(k-1)!\,[(n-1)-(k-1)]!} + \frac{(n-1)!}{k!\,[(n-1)-k]!}
    \]
    Simplificamos los factoriales en los denominadores:
    \[
    = \frac{(n-1)!}{(k-1)!\,(n-k)!} + \frac{(n-1)!}{k!\,(n-k-1)!}
    \]
    Escribimos $(n-k)! = (n-k)(n-k-1)!$ en el primer término y $k! = k(k-1)!$ en el segundo, para sacar como factor común $\dfrac{(n-1)!}{(k-1)!\,(n-k-1)!}$:
    \[
    = \frac{(n-1)!}{(k-1)!\,(n-k-1)!}\left(\frac{1}{n-k} + \frac{1}{k}\right)
    \]
    Sumamos las fracciones dentro del paréntesis, buscando el común denominador $k(n-k)$:
    \[
    = \frac{(n-1)!}{(k-1)!\,(n-k-1)!} \cdot \frac{k+(n-k)}{k(n-k)}
    \]
    Simplificamos el numerador ($k + n - k = n$):
    \[
    = \frac{(n-1)!}{(k-1)!\,(n-k-1)!} \cdot \frac{n}{k(n-k)}
    \]
    Reagrupamos usando que $(n-1)!\cdot n = n!$, $(k-1)!\cdot k = k!$ y $(n-k-1)!\cdot(n-k) = (n-k)!$:
    \[
    = \frac{n!}{k!\,(n-k)!}
    \]
    Y por definición de número combinatorio, obtenemos el resultado final:
    \[
    = \binom{n}{k}
    \]
\end{demostracion}


\begin{proposicion}[title= Teorema del Binomio]
    Sean $a, b \in \mathbb{R}$ y $n \in \mathbb{N}_0$. Entonces:
    \[
    (a+b)^n = \sum_{k=0}^{n} \binom{n}{k} a^{n-k} b^k
    \]
\end{proposicion}

\begin{demostracion}
    Procedemos por inducción en $n$.

    \textbf{Caso base ($n=0$):} $(a+b)^0 = 1$ y $\sum_{k=0}^{0} \binom{0}{k} a^{0-k}b^k = \binom{0}{0}a^0b^0 = 1$. Se cumple.

    \textbf{Paso inductivo:} Supongamos que la fórmula vale para $n$, es decir:
    \[
    (a+b)^n = \sum_{k=0}^{n} \binom{n}{k} a^{n-k} b^k
    \]
    Queremos ver que vale para $n+1$. Multiplicamos ambos lados por $(a+b)$:
    \[
    (a+b)^{n+1} = (a+b)\sum_{k=0}^{n} \binom{n}{k} a^{n-k} b^k = \sum_{k=0}^{n} \binom{n}{k} a^{n-k+1} b^k + \sum_{k=0}^{n} \binom{n}{k} a^{n-k} b^{k+1}
    \]
    Reacomodando índices para que ambas sumas queden en términos de $a^{n+1-k}b^k$, y usando la identidad de Pascal (Stifel) $\binom{n}{k-1}+\binom{n}{k} = \binom{n+1}{k}$, se obtiene:
    \[
    (a+b)^{n+1} = \sum_{k=0}^{n+1} \binom{n+1}{k} a^{n+1-k} b^k
    \]
    Lo cual completa el paso inductivo, y por lo tanto la fórmula vale para todo $n \in \mathbb{N}_0$.
\end{demostracion}
